class SmartChargingSimulation:
    """
    Smart charging with DYNAMIC REROUTING
    When SOC < 30%, calculates route to nearest station and reroutes vehicle
    """
    
    SOC_THRESHOLD = 30.0
    # SOC_TARGET = 80.0
    SOC_TARGET = 60.0
    
    # Charging stations with their edge locations
    CHARGING_STATIONS = {
        'pa_2': {
            'lane': 'E0_0',
            'edge': 'E0',
            'startPos': 745.11,
            'power': 22000,
            'efficiency': 0.95,

        },
        'pa_3': {

        },
        'pa_6': {

            'capacity': 4
        },
        'pa_7': {

            'power': 22000,
            'efficiency': 0.95,
            'capacity': 4
        },
        'pa_8': {

            'edge': 'E7',
            'startPos': 830.91,
            'power': 22000,
            'efficiency': 0.95,
            'capacity': 4

    def __init__(self, sumocfg='Test1.sumocfg', output_folder='simulation_outputs'):
        self.sumocfg = sumocfg
        self.output_folder = output_folder
        
        if not os.path.exists(output_folder):
            os.makedirs(output_folder)
        
        self.vehicle_data = []
        self.charging_events = []
        self.low_soc_alerts = []
        self.vehicle_states = {}
        self.initial_soc = {}
        self.station_positions = {}
        
        print(f"✓ Smart Charging Simulation with DYNAMIC REROUTING")
        print(f"  SOC Threshold: {self.SOC_THRESHOLD}%")
        print(f"  Target SOC: {self.SOC_TARGET}%")
        print(f"  Charging Stations: {len(self.CHARGING_STATIONS)}")
    
    def sanitize_lane_id(self, lane_id):
        """
        Sanitize lane/road IDs to prevent Excel #NAME? and formula issues.
        - Internal lanes ":" → "Internal_*"
        - Reverse edges "-" → "NEG_*"
        """
        if not lane_id:
            return "UNKNOWN"
        
        lane_id = str(lane_id)

        # Fix internal lanes (junctions)
        if lane_id.startswith(':'):
            return f"Internal_{lane_id[1:]}"

        # Fix negative edges/lane IDs that Excel thinks are formulas
        if lane_id.startswith('-'):
            return f"NEG_{lane_id[1:]}"
        
        return lane_id

    
    def get_lane_position(self, lane_id, pos):
        """Get x,y coordinates from lane and position"""
        try:
            shape = traci.lane.getShape(lane_id)
            if len(shape) >= 2:
                lane_length = traci.lane.getLength(lane_id)
                ratio = min(pos / lane_length, 1.0) if lane_length > 0 else 0
                x = shape[0][0] + ratio * (shape[-1][0] - shape[0][0])
                y = shape[0][1] + ratio * (shape[-1][1] - shape[0][1])
                return (x, y)
            return shape[0] if shape else (0, 0)
        except:
            return (0, 0)
    
    def calculate_distance(self, pos1, pos2):
        """Calculate Euclidean distance"""
        return math.sqrt((pos1[0] - pos2[0])**2 + (pos1[1] - pos2[1])**2)
    
    def find_reachable_charging_station(self, veh_id, vehicle_position):
        """
        Find nearest charging station that vehicle can reach via routing
        Tests each station to see if a route exists
        """
        current_edge = traci.vehicle.getRoadID(veh_id)
        
        # Skip if on internal edge (junction)
        if current_edge.startswith(':'):
            return None
        
        best_station = None
        best_route_length = float('inf')
        
        for station_id, station_info in self.CHARGING_STATIONS.items():
            station_edge = station_info['edge']
            
            try:
                # Try to find route from current edge to station edge
                route = traci.simulation.findRoute(current_edge, station_edge)
                
                if route and route.edges:
                    route_length = route.length
                    
                    # Choose station with shortest route
                    if route_length < best_route_length:
                        best_route_length = route_length
                        best_station = {
                            'id': station_id,
                            'info': station_info,
                            'route': route.edges,
                            'length': route_length
                        }
            except:
                continue
        
        return best_station
    
    def get_vehicle_battery_info(self, veh_id):
        """Get battery information"""
        try:
            actual_battery = float(traci.vehicle.getParameter(veh_id, "device.battery.actualBatteryCapacity"))
            max_battery = float(traci.vehicle.getParameter(veh_id, "device.battery.maximumBatteryCapacity"))
            energy_consumed = float(traci.vehicle.getParameter(veh_id, "device.battery.totalEnergyConsumed"))
            
            soc_percent = (actual_battery / max_battery * 100) if max_battery > 0 else 0
            
            return {
                'actual_battery_Wh': actual_battery,
                'max_battery_Wh': max_battery,
                'soc_percent': soc_percent,
                'energy_consumed_Wh': energy_consumed
            }
        except:
            return None
    
    def reroute_to_charging_station(self, veh_id, station_data, battery_info):
        """
        CRITICAL FIX: Reroute vehicle to charging station
        1. Set new route to station edge
        2. Add parking area stop once on correct route
        """
        
        try:
            station_id = station_data['id']
            station_info = station_data['info']
            new_route = station_data['route']
            
            # Calculate charging duration
            current_soc = battery_info['soc_percent']
            max_battery = battery_info['max_battery_Wh']
            energy_needed = (self.SOC_TARGET - current_soc) / 100 * max_battery
            charging_power = station_info['power']
            efficiency = station_info['efficiency']
            charging_duration = (energy_needed / (charging_power * efficiency)) * 3600
            charging_duration = max(60, min(charging_duration, 600))
            
            # STEP 1: Change vehicle route to go to station
            traci.vehicle.setRoute(veh_id, new_route)
            
            # STEP 2: Set parking area stop (now it's on the route!)
            traci.vehicle.setParkingAreaStop(
                veh_id,
                station_id,
                duration=charging_duration
            )
            
            print(f"🔌 REROUTED {veh_id} to {station_id}")
            print(f"   New route: {' → '.join(new_route[:3])}... ({len(new_route)} edges)")
            print(f"   Will charge for {charging_duration:.0f}s")
            print(f"   Target SOC: {current_soc:.1f}% → {self.SOC_TARGET}%")
            
            return True
            
        except Exception as e:
            print(f"⚠️ Rerouting failed for {veh_id}: {e}")
            return False
    
    def check_charging_needed(self, veh_id, battery_info, simulation_time):
        """Check if vehicle needs charging and reroute if necessary"""
        
        soc = battery_info['soc_percent']
        
        if soc < self.SOC_THRESHOLD:
            
            # Only route once per vehicle
            if veh_id in self.vehicle_states and 'routed_to_station' in self.vehicle_states[veh_id]:
                return False
            
            position = traci.vehicle.getPosition(veh_id)
            veh_type = traci.vehicle.getTypeID(veh_id)
            
            # Find REACHABLE charging station (not just nearest!)
            station_data = self.find_reachable_charging_station(veh_id, position)
            
            if not station_data:
                # print(f"⚠️ No reachable charging station for {veh_id}")
                return False
            
            soc_deficit = self.SOC_THRESHOLD - soc
            
            # Record alert
            alert = {
                'timestep_sec': simulation_time,
                'vehicle_id': veh_id,
                'vehicle_type': veh_type,
                'current_soc_percent': round(soc, 2),
                'threshold_soc_percent': self.SOC_THRESHOLD,
                'soc_below_threshold': round(soc_deficit, 2),
                'nearest_station': station_data['id'],
                'route_length_m': round(station_data['length'], 2),
                'battery_capacity_Wh': battery_info['actual_battery_Wh'],
                'max_battery_Wh': battery_info['max_battery_Wh'],
                'action': 'REROUTED_TO_STATION'
            }
            self.low_soc_alerts.append(alert)
            
            # REROUTE VEHICLE
            success = self.reroute_to_charging_station(veh_id, station_data, battery_info)
            
            if success:
                if veh_id not in self.vehicle_states:
                    self.vehicle_states[veh_id] = {}
                self.vehicle_states[veh_id]['routed_to_station'] = station_data['id']
                self.vehicle_states[veh_id]['alert_time'] = simulation_time
            
            print(f"⚠️ LOW SOC: {veh_id} ({veh_type})")
            print(f"    Current SOC: {soc:.2f}% (Threshold: {self.SOC_THRESHOLD}%)")
            print(f"    Deficit: {soc_deficit:.2f}%")
            print(f"    Rerouted to: {station_data['id']} ({station_data['length']:.0f}m route)")
            
            return True
        
        return False
    
    def monitor_real_charging(self, simulation_time):
        """Monitor actual charging events"""
        
        vehicle_ids = traci.vehicle.getIDList()
        
        for veh_id in vehicle_ids:
            try:
                charging_station_id = traci.vehicle.getParameter(veh_id, "device.battery.chargingStationId")
                
                if charging_station_id and charging_station_id != "NULL" and charging_station_id != "":
                    # Charging started
                    battery_info = self.get_vehicle_battery_info(veh_id)
                    if not battery_info:
                        continue
                    
                    if veh_id not in self.vehicle_states:
                        self.vehicle_states[veh_id] = {}
                    
                    if 'charging_start_time' not in self.vehicle_states[veh_id]:
                        self.vehicle_states[veh_id]['charging_start_time'] = simulation_time
                        self.vehicle_states[veh_id]['soc_at_start'] = battery_info['soc_percent']
                        self.vehicle_states[veh_id]['charging_station_active'] = charging_station_id
                        
                        print(f"🔌 CHARGING STARTED: {veh_id} at {charging_station_id}")
                        print(f"    SOC: {battery_info['soc_percent']:.2f}%")
                
                else:
                    # Charging completed
                    if veh_id in self.vehicle_states and 'charging_start_time' in self.vehicle_states[veh_id]:
                        battery_info = self.get_vehicle_battery_info(veh_id)
                        if not battery_info:
                            continue
                        
                        charge_start = self.vehicle_states[veh_id]['charging_start_time']
                        soc_start = self.vehicle_states[veh_id]['soc_at_start']
                        soc_end = battery_info['soc_percent']
                        duration = simulation_time - charge_start
                        station_used = self.vehicle_states[veh_id].get('charging_station_active', 'UNKNOWN')
                        
                        charging_event = {
                            'vehicle_id': veh_id,
                            'vehicle_type': traci.vehicle.getTypeID(veh_id),
                            'charging_station': station_used,
                            'start_time_sec': charge_start,
                            'end_time_sec': simulation_time,
                            'duration_sec': round(duration, 1),
                            'soc_at_start_percent': round(soc_start, 2),
                            'soc_at_end_percent': round(soc_end, 2),
                            'soc_gained_percent': round(soc_end - soc_start, 2),
                            'charging_type': 'WIRED_PLUG_IN',
                            'charging_successful': 'YES' if soc_end > soc_start else 'NO'
                        }
                        self.charging_events.append(charging_event)
                        
                        print(f"✅ CHARGING COMPLETE: {veh_id}")
                        print(f"    Duration: {duration:.1f}s")
                        print(f"    SOC: {soc_start:.2f}% → {soc_end:.2f}% (+{soc_end - soc_start:.2f}%)")
                        
                        del self.vehicle_states[veh_id]['charging_start_time']
                        del self.vehicle_states[veh_id]['soc_at_start']
                        if 'charging_station_active' in self.vehicle_states[veh_id]:
                            del self.vehicle_states[veh_id]['charging_station_active']
            
            except:
                continue
    
    def collect_vehicle_data(self, simulation_time):
        """Collect vehicle data with FIXED lane ID and column ordering"""
        
        self.monitor_real_charging(simulation_time)
        
        vehicle_ids = traci.vehicle.getIDList()
        
        for veh_id in vehicle_ids:
            try:
                # Get vehicle info
                veh_type = traci.vehicle.getTypeID(veh_id)
                speed = traci.vehicle.getSpeed(veh_id)
                position = traci.vehicle.getPosition(veh_id)
                distance = traci.vehicle.getDistance(veh_id)
                
                # Get lane and road IDs (RAW)
                lane_id_raw = traci.vehicle.getLaneID(veh_id)
                road_id_raw = traci.vehicle.getRoadID(veh_id)
                
                # Get battery info
                battery_info = self.get_vehicle_battery_info(veh_id)
                if battery_info is None:
                    continue
                
                # CRITICAL FIX: Sanitize lane and road IDs for Excel AFTER getting battery info
                lane_id = self.sanitize_lane_id(lane_id_raw)
                road_id = self.sanitize_lane_id(road_id_raw)
                
                # Track initial SOC
                if veh_id not in self.initial_soc:
                    self.initial_soc[veh_id] = battery_info['soc_percent']
                
                # Check if charging needed
                self.check_charging_needed(veh_id, battery_info, simulation_time)
                
                # Check if currently charging
                is_charging = False
                try:
                    charging_station_id = traci.vehicle.getParameter(veh_id, "device.battery.chargingStationId")
                    if charging_station_id and charging_station_id != "NULL" and charging_station_id != "":
                        is_charging = True
                except:
                    pass
                
                # CRITICAL FIX: Create record with explicit column order
                record = {
                    'timestep_sec': simulation_time,
                    'vehicle_id': veh_id,
                    'vehicle_type': veh_type,
                    'actual_battery_Wh': round(battery_info['actual_battery_Wh'], 2),
                    'max_battery_Wh': round(battery_info['max_battery_Wh'], 2),
                    'soc_percent': round(battery_info['soc_percent'], 2),
                    'initial_soc_percent': round(self.initial_soc[veh_id], 2),
                    'soc_drop_percent': round(self.initial_soc[veh_id] - battery_info['soc_percent'], 2),
                    'below_threshold': 'YES' if battery_info['soc_percent'] < self.SOC_THRESHOLD else 'NO',
                    'threshold_deficit_percent': round(self.SOC_THRESHOLD - battery_info['soc_percent'], 2) if battery_info['soc_percent'] < self.SOC_THRESHOLD else 0,
                    'energy_consumed_Wh': round(battery_info['energy_consumed_Wh'], 2),
                    'currently_charging': 'YES' if is_charging else 'NO',
                    'speed_ms': round(speed, 2),
                    'x_position': round(position[0], 2),
                    'y_position': round(position[1], 2),
                    'lane': lane_id,  # FIXED: Sanitized lane ID
                    'road_id': road_id,  # FIXED: Sanitized road ID
                    'distance_m': round(distance, 2)
                }
                
                self.vehicle_data.append(record)
                
            except Exception as e:
                # Uncomment for debugging
                # print(f"Error collecting data for {veh_id}: {e}")
                continue
    
    def run_simulation(self, gui=False, max_time=600):
        """Run simulation"""
        
        print("\n" + "="*70)
        print("SMART CHARGING WITH DYNAMIC REROUTING - EXCEL-COMPATIBLE VERSION")
        print("="*70)
        print(f"Configuration: {self.sumocfg}")
        print(f"GUI Mode: {'Enabled' if gui else 'Disabled'}")
        print(f"Max Time: {max_time} seconds")
        print("="*70 + "\n")
        
        sumo_binary = "sumo-gui" if gui else "sumo"
        sumo_cmd = [sumo_binary, "-c", self.sumocfg, "--start", "--quit-on-end"]
        
        try:
            traci.start(sumo_cmd)
            print("✓ TraCI connection established\n")
            
            simulation_time = 0
            
            while traci.simulation.getMinExpectedNumber() > 0 and simulation_time < max_time:
                traci.simulationStep()
                simulation_time = traci.simulation.getTime()
                
                if int(simulation_time) == simulation_time:
                    self.collect_vehicle_data(simulation_time)
                
                if int(simulation_time) % 30 == 0 and simulation_time > 0:
                    active = len(traci.vehicle.getIDList())
                    charging_now = sum(1 for v in self.vehicle_states.values() if 'charging_start_time' in v)
                    print(f"⏱️ Time: {int(simulation_time)}s | Vehicles: {active} | Charging: {charging_now} | Alerts: {len(self.low_soc_alerts)}")
            
            print(f"\n✓ Simulation completed at {simulation_time}s")
            traci.close()
            return True
            
        except Exception as e:
            print(f"\n✗ ERROR: {e}")
            import traceback
            traceback.print_exc()
            try:
                traci.close()
            except:
                pass
            return False
    
    def export_results(self):
        """Export results to CSV with explicit column ordering"""
        
        print("\n" + "="*70)
        print("EXPORTING RESULTS")
        print("="*70)
        
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        
        if self.vehicle_data:
            # CRITICAL FIX: Define explicit column order
            columns = [
                'timestep_sec', 'vehicle_id', 'vehicle_type',
                'actual_battery_Wh', 'max_battery_Wh', 'soc_percent',
                'initial_soc_percent', 'soc_drop_percent',
                'below_threshold', 'threshold_deficit_percent',
                'energy_consumed_Wh', 'currently_charging',
                'speed_ms', 'x_position', 'y_position',
                'lane', 'road_id', 'distance_m'
            ]
            
            df = pd.DataFrame(self.vehicle_data, columns=columns)
            filename = os.path.join(self.output_folder, f'vehicle_soc_tracking_{timestamp}.csv')
            df.to_csv(filename, index=False)
            print(f"\n✓ Vehicle tracking: {filename}")
            print(f"  Records: {len(df)}")
            print(f"  Vehicles: {df['vehicle_id'].nunique()}")
        
        if self.low_soc_alerts:
            df = pd.DataFrame(self.low_soc_alerts)
            filename = os.path.join(self.output_folder, f'low_soc_alerts_{timestamp}.csv')
            df.to_csv(filename, index=False)
            print(f"✓ Low SOC alerts: {filename}")
            print(f"  Total alerts: {len(df)}")
            print(f"  Unique vehicles: {df['vehicle_id'].nunique()}")
        
        if self.charging_events:
            df = pd.DataFrame(self.charging_events)
            filename = os.path.join(self.output_folder, f'charging_events_{timestamp}.csv')
            df.to_csv(filename, index=False)
            print(f"✓ Charging events: {filename}")
            print(f"  Total sessions: {len(df)}")
            print(f"  Avg duration: {df['duration_sec'].mean():.1f}s")
            print(f"  Avg SOC gain: {df['soc_gained_percent'].mean():.2f}%")
        
        print("="*70)


def main():
    """Main execution"""
    
    print("\n" + "="*70)
    print("SMART EV CHARGING - EXCEL-COMPATIBLE VERSION")
    print("="*70 + "\n")
    
    SUMOCFG = 'Test1.sumocfg'
    OUTPUT_FOLDER = 'simulation_outputs'
    USE_GUI = True
    MAX_TIME = 600
    
    if not os.path.exists(SUMOCFG):
        print(f"✗ ERROR: {SUMOCFG} not found!")
        return
    
    sim = SmartChargingSimulation(
        sumocfg=SUMOCFG,
        output_folder=OUTPUT_FOLDER
    )
    
    success = sim.run_simulation(gui=USE_GUI, max_time=MAX_TIME)
    
    if not success:
        print("\n✗ Simulation failed!")
        return
    
    sim.export_results()
    
    print("\n" + "="*70)
    print("✓ SIMULATION COMPLETE!")
    print(f"Output folder: {OUTPUT_FOLDER}/")
    print("="*70 + "\n")


if __name__ == "__main__":
    main()
